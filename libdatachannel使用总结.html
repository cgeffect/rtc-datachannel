libdatachannel 介绍

libdatachannel 是一个 C++ 实现的 WebRTC 库，实现了传输和rtp包的解析和拆分，但是 C++ 使用起来是比较复杂的，所以他又提供了 C 语言的 API 函数。他并不是用 C 语言把整个项目重写了，而是用 C 语言包装一下 C++ 的函数，然后提供给上层使用。

整个 cpi 封装的实现都在 capi.cpp 文件里面`

组帧

void H264RtpDepacketizer::incoming(message_vector &messages, const message_callback &) {
    while (mRtpBuffer.size() != 0) {
		uint8_t payload_type = 0;
		uint32_t current_timestamp = 0;
		size_t packets_in_timestamp = 0;

		for (const auto &pkt : mRtpBuffer) {
			auto p = reinterpret_cast<const rtc::RtpHeader *>(pkt->data());
			if (current_timestamp == 0) {
				current_timestamp = p->timestamp();
				payload_type = p->payloadType(); // should all be the same for data of the same codec
			} else if (current_timestamp != p->timestamp()) {//时间变化,新的一帧开始
				break;
			}
			packets_in_timestamp++;
		}

		if (packets_in_timestamp == mRtpBuffer.size()) {
			break;
		}
		auto begin = mRtpBuffer.begin();
		auto end = mRtpBuffer.begin() + (packets_in_timestamp - 1);
         //开始组帧，返回NALU单元
		auto frames = buildFrames(begin, end + 1, payload_type, current_timestamp);
        //放到messages中
		messages.insert(messages.end(), frames.begin(), frames.end());
		mRtpBuffer.erase(mRtpBuffer.begin(), mRtpBuffer.begin() + packets_in_timestamp);
	}
}

message_vector H264RtpDepacketizer::buildFrames(message_vector::iterator begin,end, 
                                                uint8_t payloadType, uint32_t timestamp) {
	//whb:帧的包数和最后一个包的序列号
	int packetCout = 0;
	uint16_t seqNum = 0; 
    for (auto it = begin; it != end; ++it) {
		auto pkt = it->get();
		auto pktParsed = reinterpret_cast<const rtc::RtpHeader *>(pkt->data());
		auto rtpHeaderSize = pktParsed->getSize() + pktParsed->getExtensionHeaderSize();
		auto rtpPaddingSize = 0;

		//whb:更新计数和seqNum
		packetCout++;
		seqNum = pktParsed->seqNumber();
         //解析NALU头信息
		auto nalUnitHeader = NalUnitHeader{std::to_integer<uint8_t>(pkt->at(rtpHeaderSize))};
		if (nalUnitHeader.unitType() == naluTypeFUA) {
		} else if (nalUnitHeader.unitType() > 0 && nalUnitHeader.unitType() < 24) {
			addSeparator(accessUnit);//nalu分割符
			accessUnit.insert(accessUnit.end(), pkt->begin() + rtpHeaderSize, pkt->end());
		} else if (nalUnitHeader.unitType() == naluTypeSTAPA) {//多个NALU
		} else {
			throw std::runtime_error("Unknown H264 RTP Packetization");
		}
	}

	if (!accessUnit.empty()) {
		out.emplace_back(
		    make_message(std::move(accessUnit), Message::Binary, 0, nullptr, frameInfo));
	}
}

问题：没有对rtp包做顺序检测，可能会乱序，没有做帧的完整性判断，没有时间层和空间层的处理

Nack处理

实现了应答处理，没有丢包发送

RtcpNackResponder:
  RtcpNackResponder::incoming 处理nack请求
  RtcpNackResponder::outgoing 发送的消息做缓存
RtcpReceivingSession：发送反馈消息:pli，remb，RR消息，没有nack消息

如果支持Nack，需要参考webrtc的NackRequester

拆帧发送rtp

void H264RtpPacketizer::outgoing(message_vector &messages, [[maybe_unused]] const message_callback &send) {
	message_vector result;
	for(const auto &message : messages) {
		auto nalus = splitMessage(message); //拆帧
		auto fragments = nalus->generateFragments(maxFragmentSize);

		for (size_t i = 0; i < fragments.size() - 1; i++)
			result.push_back(packetize(fragments[i], false));//组rtp包        
		result.push_back(packetize(fragments[fragments.size() - 1], true));//组最后一个rtp包
	}
	messages.swap(result);
}

RtpPacketizer::packetize(shared_ptr<binary> payload, bool mark){
    auto message = make_message(RtpHeaderSize + rtpExtHeaderSize + payload->size());
	auto *rtp = (RtpHeader *)message->data();
	rtp->setPayloadType(rtpConfig->payloadType);
	rtp->setSeqNumber(rtpConfig->sequenceNumber++); // increase sequence number
	rtp->setTimestamp(rtpConfig->timestamp);//设置时间戳
	rtp->setSsrc(rtpConfig->ssrc);
    if (mark) {//最后一个包
		rtp->setMarker(true);
	}
    //处理扩展信息
    
    //数据copy
	std::memcpy(message->data() + RtpHeaderSize + rtpExtHeaderSize, payload->data(), payload->size());
	return message;
}

时间戳的生成

last_audio_timestamp(0), last_video_timestamp(0)//初始化为0,用于计时和同步时间
void WHIPOutput::Data(bool bAudio, bool bKeyFrame, const uint8_t* pData, uint32_t length,  ts)
{
		int64_t duration = ts - last_audio_timestamp;//时间差
		Send((void*)pData, length, duration, audio_track,
		     audio_sr_reporter);
		last_audio_timestamp = ts;//更新发送时间
}

void WHIPOutput::Send(void *data, uintptr_t size, uint64_t duration, track, rtcp_sr_reporter)
{
	//Sample time is in microseconds, we need to convert it to seconds
	auto elapsed_seconds = double(duration) / (1000.0 * 1000.0);
	//转化为时钟速率时间
	uint32_t elapsed_timestamp = rtp_config->secondsToTimestamp(elapsed_seconds);
	//更新时钟速率时间
	rtp_config->timestamp = rtp_config->timestamp + elapsed_timestamp;
}

rtc::OpusRtpPacketizer::DefaultClockRate = 48000;
rtc::H264RtpPacketizer::defaultClockRate = 90 * 1000;

uint32_t RtpPacketizationConfig::getTimestampFromSeconds(double seconds, uint32_t clockRate) {
	return uint32_t(int64_t(round(seconds * double(clockRate)))); 
}
uint32_t RtpPacketizationConfig::secondsToTimestamp(double seconds) {
	return RtpPacketizationConfig::getTimestampFromSeconds(seconds, clockRate);
}

总结

如果只是简单的推RTC流，可以使用libdatachannel，如果拉流，不建议使用，除非只是简单的看一下效果，做个测试。
