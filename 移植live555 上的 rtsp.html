一 代码实例

一、V4L2视频采集模块（完整示例） 

#include <linux/videodev2.h>
#include <sys/ioctl.h>
#include <fcntl.h>

// 初始化V4L2摄像头
int init_v4l2_camera(const char* dev_path, int width, int height) {
    int fd = open(dev_path, O_RDWR);
    if (fd < 0) {
        perror("Failed to open V4L2 device");
        return -1;
    }

    // 设置视频格式（YUV422）
    struct v4l2_format fmt = {0};
    fmt.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
    fmt.fmt.pix.width = width;
    fmt.fmt.pix.height = height;
    fmt.fmt.pix.pixelformat = V4L2_PIX_FMT_YUYV; // 海思可能要求NV12
    fmt.fmt.pix.field = V4L2_FIELD_NONE;

    if (ioctl(fd, VIDIOC_S_FMT, &fmt) < 0) {
        perror("Failed to set video format");
        close(fd);
        return -1;
    }

    // 申请视频缓冲区（MMAP方式）
    struct v4l2_requestbuffers req = {0};
    req.count = 4;
    req.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
    req.memory = V4L2_MEMORY_MMAP;

    if (ioctl(fd, VIDIOC_REQBUFS, &req) < 0) {
        perror("Failed to request buffers");
        close(fd);
        return -1;
    }

    // 映射缓冲区到用户空间
    struct v4l2_buffer buf = {0};
    for (int i = 0; i < req.count; ++i) {
        buf.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
        buf.memory = V4L2_MEMORY_MMAP;
        buf.index = i;
        if (ioctl(fd, VIDIOC_QUERYBUF, &buf) < 0) {
            perror("Failed to query buffer");
            close(fd);
            return -1;
        }
        void* buffer = mmap(NULL, buf.length, PROT_READ | PROT_WRITE, MAP_SHARED, fd, buf.m.offset);
        // 保存buffer指针...
    }

    // 启动视频流
    enum v4l2_buf_type type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
    if (ioctl(fd, VIDIOC_STREAMON, &type) < 0) {
        perror("Failed to start streaming");
        close(fd);
        return -1;
    }

    return fd; // 返回设备句柄
}

关键点解释 ：

  • 海思平台可能要求输入NV12格式，需根据硬件文档调整 V4L2_PIX_FMT_*
  • MMAP方式减少内存拷贝，适合高帧率场景
  • 需循环调用 VIDIOC_QBUF 和 VIDIOC_DQBUF 获取视频帧

二、海思H.264硬件编码模块 

#include "hi_comm_venc.h"
#include "mpi_venc.h"

// 初始化海思编码通道
HI_S32 init_hisi_venc(int width, int height, int fps) {
    HI_S32 s32Ret;
    VENC_CHN VencChn = 0;
    VENC_ATTR_S stVencAttr;

    // 配置编码参数
    memset(&stVencAttr, 0, sizeof(VENC_ATTR_S));
    stVencAttr.enType = PT_H264;
    stVencAttr.stRcAttr.enRcMode = VENC_RC_MODE_H264CBR;
    stVencAttr.stRcAttr.stH264Cbr.u32Gop = fps * 2; // GOP为2秒
    stVencAttr.stRcAttr.stH264Cbr.u32BitRate = 2000; // 2000kbps
    stVencAttr.stVencAttr.stAttrH264e.u32MaxPicWidth = width;
    stVencAttr.stVencAttr.stAttrH264e.u32MaxPicHeight = height;
    stVencAttr.stVencAttr.stAttrH264e.u32PicWidth = width;
    stVencAttr.stVencAttr.stAttrH264e.u32PicHeight = height;

    // 创建编码通道
    s32Ret = HI_MPI_VENC_CreateChn(VencChn, &stVencAttr);
    if (s32Ret != HI_SUCCESS) {
        printf("Create VENC channel failed: 0x%x\n", s32Ret);
        return s32Ret;
    }

    // 设置帧率
    VENC_PARAM_FRMRATE_PARA_S stFrmRate;
    stFrmRate.stSrcFrmRate.s32FrmRate = fps * 1000; // 以毫秒为单位
    stFrmRate.stDstFrmRate.s32FrmRate = fps * 1000;
    HI_MPI_VENC_SetFrmRate(VencChn, &stFrmRate);

    return HI_SUCCESS;
}

// 获取编码后的H.264流
void get_h264_stream(VENC_CHN VencChn, unsigned char** ppBuffer, HI_U32* pSize) {
    VENC_STREAM_S stStream;
    HI_S32 s32Ret = HI_MPI_VENC_GetStream(VencChn, &stStream, 2000); // 2秒超时
    if (s32Ret == HI_SUCCESS) {
        *ppBuffer = stStream.pstPack[0].pu8Addr;
        *pSize = stStream.pstPack[0].u32Len;
        HI_MPI_VENC_ReleaseStream(VencChn, &stStream); // 必须释放
    }
}

关键点 ：

  • 使用 HI_MPI_VENC_GetStream 非阻塞方式获取码流
  • 必须调用 HI_MPI_VENC_ReleaseStream 释放资源，否则会导致内存泄漏
  • 海思SDK要求严格的内存对齐，需按照文档配置参数

三、ALSA音频采集与AAC编码 

#include <alsa/asoundlib.h>
#include <fdk-aac/aacenc_lib.h>

// 初始化ALSA音频采集
snd_pcm_t* init_alsa(const char* device, int sample_rate, int channels) {
    snd_pcm_t* handle;
    snd_pcm_hw_params_t* params;

    if (snd_pcm_open(&handle, device, SND_PCM_STREAM_CAPTURE, 0) < 0) {
        fprintf(stderr, "ALSA open error\n");
        return NULL;
    }

    snd_pcm_hw_params_malloc(&params);
    snd_pcm_hw_params_any(handle, params);

    // 设置参数
    snd_pcm_hw_params_set_access(handle, params, SND_PCM_ACCESS_RW_INTERLEAVED);
    snd_pcm_hw_params_set_format(handle, params, SND_PCM_FORMAT_S16_LE);
    snd_pcm_hw_params_set_channels(handle, params, channels);
    snd_pcm_hw_params_set_rate_near(handle, params, (unsigned int*)&sample_rate, 0);

    // 应用参数
    if (snd_pcm_hw_params(handle, params) < 0) {
        fprintf(stderr, "ALSA set params failed\n");
        snd_pcm_close(handle);
        return NULL;
    }

    return handle;
}

// 初始化AAC编码器
AACENCODER* init_aac_encoder(int sample_rate, int channels) {
    AACENCODER* hEncoder;
    if (aacEncOpen(&hEncoder, 0, channels) != AACENC_OK) {
        return NULL;
    }

    // 设置编码参数
    aacEncoder_SetParam(hEncoder, AACENC_AOT, AOT_AAC_LC);
    aacEncoder_SetParam(hEncoder, AACENC_SAMPLERATE, sample_rate);
    aacEncoder_SetParam(hEncoder, AACENC_CHANNELMODE, MODE_2);
    aacEncoder_SetParam(hEncoder, AACENC_BITRATE, 64000); // 64kbps

    if (aacEncEncode(hEncoder, NULL, NULL, NULL, NULL) != AACENC_OK) {
        aacEncClose(&hEncoder);
        return NULL;
    }

    return hEncoder;
}

四、live555自定义FramedSource实现 

#include <liveMedia.hh>

class Hi3516VideoSource : public FramedSource {
public:
    static Hi3516VideoSource* createNew(UsageEnvironment& env, VENC_CHN vencChn) {
        return new Hi3516VideoSource(env, vencChn);
    }

protected:
    Hi3516VideoSource(UsageEnvironment& env, VENC_CHN vencChn) 
        : FramedSource(env), mVencChn(vencChn) {}

    virtual void doGetNextFrame() {
        // 从海思编码器获取H.264数据
        unsigned char* pData = nullptr;
        HI_U32 dataSize = 0;
        get_h264_stream(mVencChn, &pData, &dataSize);

        if (dataSize > 0) {
            if (dataSize > fMaxSize) { // 处理分包
                fNumTruncatedBytes = dataSize - fMaxSize;
                dataSize = fMaxSize;
            }
            memcpy(fTo, pData, dataSize);
            fFrameSize = dataSize;

            // 设置时间戳（使用系统时钟）
            gettimeofday(&fPresentationTime, NULL);
        } else {
            fFrameSize = 0;
        }

        // 立即通知框架可以发送数据
        afterGetting(this);
    }

private:
    VENC_CHN mVencChn;
};

关键逻辑 ：

  • doGetNextFrame() 是live555的核心回调函数，当需要发送数据时自动触发
  • 时间戳必须转换为RTP时间基准（视频：90000Hz，音频：采样率）
  • 若数据超过缓冲区大小，需设置 fNumTruncatedBytes

五、RTSP服务端搭建 

#include "liveMedia.hh"
#include "BasicUsageEnvironment.hh"

void start_rtsp_server() {
    //创建任务管理器环境
    TaskScheduler* scheduler = BasicTaskScheduler::createNew();
    //控制台应用环境
    UsageEnvironment* env = BasicUsageEnvironment::createNew(*scheduler);

    // 创建RTSP服务器（端口554需要root权限）
    RTSPServer* rtspServer = RTSPServer::createNew(*env, 554, NULL);
    if (rtspServer == NULL) {
        *env << "Failed to create RTSP server: " << env->getResultMsg() << "\n";
        exit(1);
    }

    // 创建媒体会话（H.264 + AAC）
    ServerMediaSession* sms = ServerMediaSession::createNew(
        *env, "live", "Hi3516 Live Stream", "Session for Hi3516");

    // 添加视频子会话
    sms->addSubsession(H264VideoFileServerMediaSubsession::createNew(
        *env, "dummy", NULL)); // 实际应使用自定义Source

    // 添加音频子会话（AAC）
    sms->addSubsession(ADTSAudioFileServerMediaSubsession::createNew(
        *env, "dummy", NULL));

    //把媒体会话添加到 RTSP服务器实例中
    //注意:Live555 的RTSPServer 在调用 addServerMediaSession() 后，自动监听客户端请求并处理流传输。所有“发布”逻辑已封装在库内部
    rtspServer->addServerMediaSession(sms);

    // 打印RTSP URL
    char* url = rtspServer->rtspURL(sms);
    *env << "RTSP URL: " << url << "\n";
    delete[] url;

    env->taskScheduler().doEventLoop(); // 进入主循环
}

// 全局时间戳生成器（单位：微秒）
uint64_t get_ntp_timestamp() {
    struct timeval tv;
    gettimeofday(&tv, NULL);
    return (uint64_t)tv.tv_sec * 1000000 + tv.tv_usec;
}

// 在FramedSource中设置时间戳
void Hi3516VideoSource::doGetNextFrame() {
    // ... 获取数据 ...
    uint64_t now = get_ntp_timestamp();
    fPresentationTime.tv_sec = now / 1000000;
    fPresentationTime.tv_usec = now % 1000000;
}

// 音频同步需转换为采样数
uint32_t audio_rtp_timestamp(uint64_t ntp_ts, int sample_rate) {
    return (uint32_t)((ntp_ts * sample_rate) / 1000000);
}

二 live555 源码分析

一 Event 处理

在 Live555 库中， env->taskScheduler().doEventLoop() 是事件循环的核心入口，负责调度和处理所有异步事件。以下是对其处理的常见事件类型、处理机制及实际场景的详细说明：

一、事件循环的核心作用

doEventLoop() 是一个无限循环，持续监听并处理以下事件：

  1. Socket I/O 事件 （网络数据到达、连接请求等）。
  2. 定时任务 （如 RTCP 报告的周期性发送）。
  3. 延迟任务 （由代码显式触发的非即时任务）。
  4. 自定义事件 （用户注册的特殊事件处理）。
二、常见事件类型及处理机制
1. Socket I/O 事件
  • 触发条件 ：套接字（Socket）有数据可读、可写或发生异常。
  • 注册方式 ：通过 envir().taskScheduler().turnOnBackgroundReadHandling() 注册读事件。
  • 处理逻辑 ：
    • 读事件 ：当客户端发送 RTSP 请求或 RTP/RTCP 数据时，回调 incomingDataHandler 解析数据。
    • 写事件 ：发送 RTP 数据包或 RTSP 响应时触发（通常由 Live555 内部管理）。
  • 示例场景 ：
    • 客户端发送 PLAY 请求 → 触发读事件 → 解析请求并启动 RTP 传输。
    • RTP 数据包准备好 → 触发写事件 → 发送数据到客户端。
2. 定时器事件（Delayed Tasks）
  • 触发条件 ：预设的时间间隔到达。
  • 注册方式 ：通过 taskScheduler().scheduleDelayedTask() 注册。
  • 处理逻辑 ：
    • 定时任务按优先级插入队列，到期后调用绑定的处理函数。
    • 任务执行后，可选择重新注册以实现周期性触发。
  • 示例场景 ：
    • RTCP 报告发送 ：每 5 秒发送一次 Receiver Report（RR）。
    • 保活检测 ：定期检查客户端连接状态。
3. 延迟任务（One-Off Tasks）
  • 触发条件 ：由代码显式触发，通常用于非实时操作。
  • 注册方式 ：通过 taskScheduler().scheduleDelayedTask(0, ...) 注册即时任务。
  • 处理逻辑 ：
    • 任务立即加入队列，在下一个事件循环迭代中执行。
  • 示例场景 ：
    • 媒体源数据就绪后，触发 RTP 打包任务。
    • 资源释放的延迟执行（避免阻塞主线程）。
4. 用户自定义事件
  • 触发条件 ：开发者通过自定义触发器（如信号量、队列）激活。
  • 注册方式 ：通过继承 TaskScheduler 或使用中间层封装。
  • 处理逻辑 ：
    • 自定义事件处理器需集成到 Live555 事件循环中。
  • 示例场景 ：
    • 从硬件设备（如摄像头）拉取数据后触发编码任务。
    • 外部系统通知流媒体参数调整（如动态码率切换）。
三、事件处理流程（代码级解析）

以下是 Live555 事件循环的简化逻辑： 

void BasicTaskScheduler::doEventLoop(char* watchVariable) {
    while (!watchVariable || *watchVariable == 0) {
        // 1. 处理已到期的定时任务
        DelayInterval timeToDelay = fDelayQueue.timeToNextAlarm();
        if (timeToDelay <= 0) {
            fDelayQueue.handleAlarm();
            continue;
        }

        // 2. 监听 Socket I/O 事件（select/poll/epoll）
        int selectResult = select(fMaxNumSockets, &fReadSet, &fWriteSet, &fExceptionSet, &tv_timeout);

        // 3. 处理读/写事件
        if (selectResult > 0) {
            for (int i = 0; i < fMaxNumSockets; ++i) {
                if (FD_ISSET(i, &fReadSet))  handleRead(i);
                if (FD_ISSET(i, &fWriteSet)) handleWrite(i);
            }
        }

        // 4. 处理其他后台任务（如信号处理）
        SingleStep(fMaxDelayTime);
    }
}
四、实际场景与调试技巧
1. 处理 RTSP 请求
  • 事件流 ：
    1. 客户端连接 → 触发 Socket 读事件 → 解析 OPTIONS / DESCRIBE 请求。
    2. 服务器生成 SDP 描述 → 发送 RTSP 响应。
    3. 客户端 SETUP → 创建 RTP/RTCP 会话。
    4. 客户端 PLAY → 启动 RTP 数据传输。
2. 发送 RTP 数据
  • 事件流 ：
    1. 媒体源（如 DeviceSource ）数据就绪 → 调用 FramedSource::afterGetting() 。
    2. 触发 doGetNextFrame() → 数据传递给 RTP 打包器。
    3. RTP 包准备好 → 触发 Socket 写事件 → 发送数据。
3. 调试事件阻塞
  • 现象 ： doEventLoop() 卡死或无响应。
  • 排查方法 ：
    • 日志分析 ：检查是否有任务未释放或死循环。
    • 超时设置 ：在 select() 调用中设置合理超时。
    • 性能分析 ：检查自定义任务是否耗时过长（应避免阻塞）。
五、自定义事件扩展

若需添加自定义事件处理器（如处理硬件中断）： 

class CustomTask : public TaskScheduler::BackgroundHandler {
public:
    static CustomTask* createNew(TaskScheduler& scheduler) {
        return new CustomTask(scheduler);
    }

    virtual void handleTimeout() {
        // 自定义事件处理逻辑（如读取硬件缓冲区）
        envir().taskScheduler().triggerEvent(&fHandler, this);
    }

private:
    CustomTask(TaskScheduler& scheduler) {
        // 注册自定义事件到调度器
        scheduler.setBackgroundHandling(fileno(stdin), SOCKET_READABLE, 
            (TaskScheduler::BackgroundHandlerProc*)&handlerProc, this);
    }

    static void handlerProc(CustomTask* task, int /*mask*/) {
        task->handleTimeout();
    }
};

// 在初始化时注册
CustomTask::createNew(*scheduler);
